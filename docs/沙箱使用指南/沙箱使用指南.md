# go-judge 沙箱 Java 接口使用指南

## 📋 接口概览

go-judge 沙箱通过 REST API 提供代码执行服务，支持完整的 Java 编译、运行和评测流程。

## 🔗 核心接口

### 1. 代码执行接口
- **URL**: `POST /run`
- **功能**: 执行 Java 代码编译和运行
- **Content-Type**: `application/json`

### 2. 文件管理接口
- **GET /file**: 列出缓存文件
- **DELETE /file/{fileId}**: 删除缓存文件
- **GET /version**: 获取版本信息

## ☕ Java 接口模板

### 基础请求结构

```java
// Java HTTP 客户端封装
public class GoJudgeClient {
    private String baseUrl;
    private HttpClient httpClient;
    
    public GoJudgeClient(String baseUrl) {
        this.baseUrl = baseUrl;
        this.httpClient = HttpClient.newBuilder()
            .timeout(Duration.ofSeconds(30))
            .build();
    }
    
    // 编译 Java 代码
    public CompileResult compileJava(String sourceCode) {
        CompileRequest request = new CompileRequest();
        request.cmd = Arrays.asList(new Command());
        
        Command cmd = request.cmd.get(0);
        cmd.args = Arrays.asList("/usr/bin/javac", "Main.java");
        cmd.env = Arrays.asList(
            "PATH=/usr/bin:/bin",
            "JAVA_HOME=/usr/lib/jvm/java-17-openjdk-amd64"
        );
        
        // 文件描述符
        cmd.files = Arrays.asList(
            new FileDescriptor("", null, null),           // stdin
            new FileDescriptor(null, "stdout", 10240),    // stdout
            new FileDescriptor(null, "stderr", 10240)     // stderr
        );
        
        // 资源限制
        cmd.cpuLimit = 10_000_000_000L;      // 10秒
        cmd.memoryLimit = 268_435_456L;      // 256MB
        cmd.procLimit = 50;
        
        // 输入文件
        cmd.copyIn = Map.of("Main.java", new InputFile(sourceCode, null));
        
        // 输出配置
        cmd.copyOut = Arrays.asList("stdout", "stderr");
        cmd.copyOutCached = Arrays.asList("Main.class");
        
        return sendRequest("/run", request, CompileResult.class);
    }
    
    // 运行 Java 程序
    public RunResult runJava(String classFileId, String input) {
        RunRequest request = new RunRequest();
        request.cmd = Arrays.asList(new Command());
        
        Command cmd = request.cmd.get(0);
        cmd.args = Arrays.asList("/usr/bin/java", "Main");
        cmd.env = Arrays.asList(
            "PATH=/usr/bin:/bin",
            "JAVA_HOME=/usr/lib/jvm/java-17-openjdk-amd64"
        );
        
        cmd.files = Arrays.asList(
            new FileDescriptor(input, null, null),        // stdin
            new FileDescriptor(null, "stdout", 10240),    // stdout
            new FileDescriptor(null, "stderr", 10240)     // stderr
        );
        
        cmd.cpuLimit = 5_000_000_000L;       // 5秒
        cmd.memoryLimit = 134_217_728L;      // 128MB
        cmd.procLimit = 1;
        
        // 使用编译生成的 class 文件
        cmd.copyIn = Map.of("Main.class", new InputFile(null, classFileId));
        cmd.copyOut = Arrays.asList("stdout", "stderr");
        
        return sendRequest("/run", request, RunResult.class);
    }
}
```

### 请求数据结构

```java
// 主请求类
public class CompileRequest {
    public List<Command> cmd;
}

// 命令配置
public class Command {
    public List<String> args;              // 执行命令
    public List<String> env;               // 环境变量
    public List<FileDescriptor> files;     // 文件描述符
    public Long cpuLimit;                  // CPU时间限制(纳秒)
    public Long memoryLimit;               // 内存限制(字节)
    public Integer procLimit;              // 进程数限制
    public Map<String, InputFile> copyIn;  // 输入文件
    public List<String> copyOut;           // 输出文件
    public List<String> copyOutCached;     // 缓存文件
}

// 文件描述符
public class FileDescriptor {
    public String content;    // 文件内容(stdin)
    public String name;       // 文件名(stdout/stderr)
    public Integer max;       // 最大字节数
    
    public FileDescriptor(String content, String name, Integer max) {
        this.content = content;
        this.name = name;
        this.max = max;
    }
}

// 输入文件
public class InputFile {
    public String content;    // 文件内容
    public String fileId;     // 文件ID(用于缓存文件)
    
    public InputFile(String content, String fileId) {
        this.content = content;
        this.fileId = fileId;
    }
}
```

## 📝 完整使用示例

```java
public class JavaJudgeExample {
    public static void main(String[] args) {
        GoJudgeClient client = new GoJudgeClient("http://localhost:5050");
        
        // Java 源代码
        String sourceCode = """
            import java.util.Scanner;
            public class Main {
                public static void main(String[] args) {
                    Scanner sc = new Scanner(System.in);
                    int a = sc.nextInt();
                    int b = sc.nextInt();
                    System.out.println(a + b);
                }
            }
            """;
        
        try {
            // 1. 编译阶段
            System.out.println("🔧 编译 Java 代码...");
            CompileResult compileResult = client.compileJava(sourceCode);
            
            if (!"Accepted".equals(compileResult.status)) {
                System.err.println("编译失败: " + compileResult.files.get("stderr"));
                return;
            }
            
            String classFileId = compileResult.fileIds.get("Main.class");
            System.out.println("✅ 编译成功，文件ID: " + classFileId);
            
            // 2. 运行阶段
            System.out.println("🚀 运行 Java 程序...");
            String input = "3 5\n";  // 测试输入
            
            RunResult runResult = client.runJava(classFileId, input);
            
            if ("Accepted".equals(runResult.status)) {
                String output = runResult.files.get("stdout");
                System.out.println("✅ 运行成功");
                System.out.println("📤 程序输出: " + output.trim());
                System.out.println("⏱️  运行时间: " + (runResult.time / 1_000_000) + " ms");
                System.out.println("🧠 内存使用: " + (runResult.memory / 1024) + " KB");
            } else {
                System.err.println("❌ 运行失败: " + runResult.status);
                if (runResult.files.containsKey("stderr")) {
                    System.err.println("错误信息: " + runResult.files.get("stderr"));
                }
            }
            
            // 3. 清理缓存文件
            client.deleteFile(classFileId);
            System.out.println("🧹 缓存文件已清理");
            
        } catch (Exception e) {
            System.err.println("请求失败: " + e.getMessage());
        }
    }
}
```

## 📊 状态码说明

| 状态 | 含义 | 处理建议 |
|------|------|----------|
| `Accepted` | 正常完成 | 继续下一步 |
| `Memory Limit Exceeded` | 内存超限 | 优化内存使用 |
| `Time Limit Exceeded` | 时间超限 | 优化算法效率 |
| `Nonzero Exit Status` | 非零退出 | 检查编译错误 |
| `Runtime Error` | 运行时错误 | 检查程序逻辑 |

## 🔧 Spring Boot 集成示例

```java
@RestController
@RequestMapping("/api/judge")
public class JudgeController {
    
    @Autowired
    private GoJudgeClient judgeClient;
    
    @PostMapping("/submit")
    public ResponseEntity<JudgeResult> submitCode(@RequestBody SubmitRequest request) {
        try {
            // 编译
            CompileResult compileResult = judgeClient.compileJava(request.getSourceCode());
            if (!"Accepted".equals(compileResult.status)) {
                return ResponseEntity.ok(JudgeResult.compileError(compileResult.files.get("stderr")));
            }
            
            // 运行测试用例
            String classFileId = compileResult.fileIds.get("Main.class");
            List<TestCaseResult> results = new ArrayList<>();
            
            for (TestCase testCase : request.getTestCases()) {
                RunResult runResult = judgeClient.runJava(classFileId, testCase.getInput());
                
                TestCaseResult tcResult = new TestCaseResult();
                tcResult.setStatus(runResult.status);
                tcResult.setExpectedOutput(testCase.getOutput());
                tcResult.setActualOutput(runResult.files.get("stdout"));
                tcResult.setTime(runResult.time / 1_000_000); // 转换为毫秒
                tcResult.setMemory(runResult.memory / 1024);  // 转换为KB
                
                results.add(tcResult);
            }
            
            // 清理文件
            judgeClient.deleteFile(classFileId);
            
            return ResponseEntity.ok(JudgeResult.success(results));
            
        } catch (Exception e) {
            return ResponseEntity.ok(JudgeResult.systemError(e.getMessage()));
        }
    }
}
```
